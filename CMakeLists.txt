cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(VirtualCameraSDK VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with Visual Studio 2022 C++ tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CAMERA_ASAN "Instrument protocol tests with AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
add_library(CameraSDK SDK/Camera.cpp)
target_include_directories(CameraSDK PUBLIC SDK)
target_link_libraries(CameraSDK PUBLIC winhttp crypt32)
add_executable(CameraStudio WIN32 App/Main.cpp)
target_link_libraries(CameraStudio PRIVATE CameraSDK user32 gdi32)
target_link_options(CameraStudio PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(CameraColorBars Examples/ColorBars.cpp)
target_link_libraries(CameraColorBars PRIVATE CameraSDK)
target_link_options(CameraColorBars PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(CameraProtocolTests Tests/ProtocolTests.cpp)
target_include_directories(CameraProtocolTests PRIVATE SDK)
if(CAMERA_ASAN)
 target_compile_options(CameraProtocolTests PRIVATE /fsanitize=address /Zi)
 target_link_options(CameraProtocolTests PRIVATE /INCREMENTAL:NO /DEBUG)
endif()
enable_testing()
add_test(NAME ProtocolValidation COMMAND CameraProtocolTests)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque signed TitanVirtualCamera INF/CAT/SYS package, optional")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 add_test(NAME SignedCatalogMembershipAndTamperRejection COMMAND CameraPackageTests "${DRIVER_PACKAGE_DIRECTORY}")
 enable_language(RC)
 foreach(name TitanVirtualCamera.inf TitanVirtualCamera.cat TitanVirtualCamera.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete driver package: ${name}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualCamera.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualCamera.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualCamera.sys\"\n")
endif()
add_library(CameraSetupCore Installer/Setup.cpp)
target_link_libraries(CameraSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32)
add_executable(CameraPackageTests Tests/PackageTests.cpp)
target_include_directories(CameraPackageTests PRIVATE Installer)
target_link_libraries(CameraPackageTests PRIVATE CameraSetupCore)
add_test(NAME PackageIdentityAndTrust COMMAND CameraPackageTests)
add_executable(VirtualCameraSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(VirtualCameraSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target VirtualCameraSetup VirtualCameraSetupConsole)
 target_link_libraries(${target} PRIVATE CameraSetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
